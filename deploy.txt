name: Deploy Migrations

on:
  push:
    branches:
      - dev
      - main
  workflow_dispatch:

jobs:
  deploy-dev:
    runs-on: ubuntu-latest

    if: github.ref_name == 'dev'
    environment: dev

    env:
      SNOWFLAKE_ACCOUNT: ${{ vars.SNOWFLAKE_ACCOUNT }}
      SNOWFLAKE_USER: ${{ vars.SNOWFLAKE_USER }}
      SNOWFLAKE_ROLE: ${{ vars.SNOWFLAKE_ROLE }}
      SNOWFLAKE_WAREHOUSE: ${{ vars.SNOWFLAKE_WAREHOUSE }}
      SNOWFLAKE_DATABASE: ${{ vars.SNOWFLAKE_DATABASE }}
      SNOWFLAKE_SCHEMA: ${{ vars.SNOWFLAKE_SCHEMA }}
      SNOWFLAKE_AUTHENTICATOR: "snowflake_jwt"

    name: Deploy to DEV
    steps:
      - name: Checkout
        uses: actions/checkout@v3

      - name: Set up Python
        uses: actions/setup-python@v4
        with:
          python-version: '3.11'

      - name: Install schemachange
        run: |
          python -m pip install --upgrade pip
          pip install schemachange

      - name: Create private key file
        run: |
          echo "${{ secrets.PRIVATE_KEY }}" > private_key.pem
          chmod 600 private_key.pem

      - name: Detect folder
        id: detect
        run: |
          FILE=$(git diff --name-only HEAD~1 HEAD | grep "stored_procs/migrations/scripts/" | head -n1)
          FOLDER=$(echo "$FILE" | cut -d'/' -f5)
          echo "folder=$FOLDER" >> $GITHUB_OUTPUT

      - name: Deploy Common
        if: ${{ steps.detect.outputs.folder == 'Common_stored_procs' }}
        run: |
          for CFG in client1-config.yml client2-config.yml client3-config.yml; do
            echo "Deploying for config: $CFG"
            schemachange deploy \
              -f "$GITHUB_WORKSPACE/stored_procs/migrations" \
              -m "$GITHUB_WORKSPACE/stored_procs/migrations/scripts" \
              --config-folder "$GITHUB_WORKSPACE/stored_procs/migrations/config" \
              --config-file-name $CFG \
              --snowflake-private-key-path private_key.pem
          done

      - name: Deploy Client1
        if: ${{ steps.detect.outputs.folder == 'Client1' }}
        run: |
          schemachange deploy \
            -f stored_procs/migrations \
            -m stored_procs/migrations/scripts \
            --config-folder stored_procs/migrations/config \
            --config-file-name client1-config.yml \
            --snowflake-private-key-path private_key.pem

      - name: Deploy Client2
        if: ${{ steps.detect.outputs.folder == 'Client2' }}
        run: |
          schemachange deploy \
            -f stored_procs/migrations \
            -m stored_procs/migrations/scripts \
            --config-folder stored_procs/migrations/config \
            --config-file-name client2-config.yml \
            --snowflake-private-key-path private_key.pem

      - name: Deploy Client3
        if: ${{ steps.detect.outputs.folder == 'Client3' }}
        run: |
          schemachange deploy \
            -f stored_procs/migrations \
            -m stored_procs/migrations/scripts \
            --config-folder stored_procs/migrations/config \
            --config-file-name client3-config.yml \
            --snowflake-private-key-path private_key.pem
